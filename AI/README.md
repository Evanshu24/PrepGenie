# Commands to run the agent:

-- First make a virtual env and activate it(optional):

python -m venv (your env name)

If on linux:

If on linux:

source ./(your env name)/bin/activate

If on windows:

.\(your env name)\Scripts\activate

-- Install all the dependencies:

pip install -r requirements.txt

-- Now create a .env file and then add your huggingface api token

-- Now run the program

python app.py
